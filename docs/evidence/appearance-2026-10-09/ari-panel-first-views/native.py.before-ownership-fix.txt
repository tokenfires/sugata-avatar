def rna(o):

 for prop in props:
  key=prop.identifier;kind=prop.type
  if key=='rna_type' or kind not in ('BOOLEAN','INT','FLOAT','STRING','ENUM'):continue
  # Filter pointer/collection before getters; STRING/ENUM have no numeric array duty.
  if kind in ('STRING','ENUM'):
   v=getattr(o,key)
   if isinstance(v,set):v=sorted(v)
   check(isinstance(v,(str,list)) and len(v)<=4096,'Bounded textual RNA '+key)
  else:
   if not hasattr(prop,'is_array') or not hasattr(prop,'array_length'):
    R['unsupported_descriptor']={'owner_type':o.bl_rna.identifier,'property':key,'descriptor_type':prop.bl_rna.identifier,'kind':kind,'is_array_supported':hasattr(prop,'is_array'),'array_length_supported':hasattr(prop,'array_length')}
    raise RuntimeError('Unsupported numeric RNA descriptor '+key)
   check(prop.array_length<=16,'Numeric RNA array bound '+key)
